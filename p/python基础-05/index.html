p1 = np.polyfit(x, f(x), 3)  # 输入1维x，y，然后选定要拟合的次数

create_plot([x] * 4, [f(x), y1, y2, y3], ['b.','r','g','c'], ['f(x)','3次','5次','7次'], ['x', 'f(x)'])  # 画出图像

X = np.hstack([np.ones(len(x)).reshape(-1,1), x.reshape(-1,1), x.reshape(-1,1)**2, x.reshape(-1,1)**3])  # 构造X矩阵
b = np.linalg.lstsq(X, f(x), rcond=None)  # np.linalg.lstsq(X, y)求解线性回归系数，这边要求输入的是设计矩阵
y_pred = X @ b[0]  # 预测值
create_plot([x] * 2, [f(x), y_pred], ['b.', 'r.'], ['f(x)', 'y_pred'], ['x', 'f(x)'])

import scipy
x = np.linspace(0,100,10)
y = np.log(x + 1) + np.random.normal(0,0.2,size=len(x)) - 3 * np.sin(x)
x_new = np.linspace(0,100,1000)
# 输入x,y 使用kind确定插值方式，返回一个函数对象
g1 = scipy.interpolate.interp1d(x,y,kind = 'linear')  # 线性插值
g2 = scipy.interpolate.interp1d(x,y,kind = 'quadratic')  # 二次插值
g3 = scipy.interpolate.interp1d(x,y,kind = 'cubic')  # 样条插值
create_plot([x, x_new, x_new, x_new], [y, g1(x_new), g2(x_new), g3(x_new)], ['b.','r-','g--','c'], ['y', 'g1(x)', 'g2(x)', 'g3(x)'], ['x', 'y'])

from scipy.interpolate import lagrange  # 导入Lagrange 插值包
x = np.linspace(0,10,10)
y = np.log(x + 1) + np.random.normal(0,0.2,size=len(x)) - 3 * np.sin(x)

ret = lagrange(x,y)  # 生成插值函数
print(ret)

x_i = np.linspace(0,10,100)
create_plot([x, x_i], [y, ret(x_i)], ['b.','r-'], ['y', 'lagrange(x)'], ['x', 'y'])

from scipy import optimize as sco
def f(p):
    x, y = p
    return np.sin(x) + np.cos(y) + x ** 2 + y ** 2

ps = sco.brute(f,((-5,5,1),(-5,5,1)))  # brute会把x,y打包成一个数组传进去，所以要在函数里解包 ((-5,5,1),(-5,5,1))表示x和y的取值范围, 1表示步长
opt1 = f(ps)

ps2 = sco.fmin(f, ps, xtol = 1e-8, ftol = 1e-8)  # fmin为梯度下降法 ps为初始猜测值，xtol和ftol分别为x和f的收敛条件
opt2 = f(ps2)

from scipy.optimize import minimize_scalar 
def f(x):
    return (x - 2) * x * (x + 2)**2

# 使用 minimize_scalar 求极小值
# method='brent' 是一种高效的标量优化算法
res = minimize_scalar(f, method='brent')

print("极小值点 x =", res.x)
print("极小值 f(x) =", res.fun)

import scipy.integrate as sci

# 基本用法：sci.quad(func, a, b)
def f(x):
    return x + 1

v, err = sci.quad(f, 1, 2)  # ∫₁² (x+1) dx  v是积分值，err是绝对误差
print(v)  # 2.5
