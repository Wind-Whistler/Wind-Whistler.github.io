Introduction

正则表达式的定义

正则表达式是一种用于描述字符串模式的特殊语法规则，它由一系列普通字符和元字符组成，能够精确地定义一组符合特定规则的文本序列。在Python中，正则表达式通过内置的re模块来实现，这个模块提供了丰富的函数和方法来处理基于模式的字符串操作。简单来说，正则表达式就像一种微型语言，让你能够用简洁的方式描述复杂的文本结构，比如匹配邮箱地址、电话号码、IP地址等具有固定格式的信息。

正则表达式的目标

正则表达式的主要目的是解决字符串处理中的模式匹配问题，它让我们能够高效地完成三类核心任务：一是验证输入数据的合法性，比如检查用户输入的邮箱是否符合格式要求；二是从大量文本中提取特定的信息，例如从网页源码中抓取所有链接；三是执行复杂的查找替换操作，比如将文档中所有日期格式统一转换。相比传统的字符串方法，正则表达式提供了一种更灵活、更强大的方式来处理非结构化文本数据，尤其适合需要处理复杂匹配逻辑的场景。

def match_pattern(pattern, test_string):  # 只要字符串里有与pattern匹配的部分即为匹配成功
    if re.search(pattern, test_string):   # re.search()可以在整个字符串中搜索匹配项
        return test_string
    else:
        return 'Fail to match'

# '.'可以匹配单个字符
match = match_pattern(r'ab.c', 'abdc')
print(match)  # 输出abdc

# '^'可以匹配字符串的开头
match = match_pattern(r'^gg', 'ggbond')
print(match)  # 输出ggbond

# '$'可以匹配字符串的结尾
match = match_pattern(r'on$', 'python')
print(match)  # 输出python

# '[]'可以匹配方括号内的任意一个字符
match = match_pattern(r'[a-z]', 'b')
print(match)  # 输出b
match = match_pattern(r'[0-9]', '1')
print(match)  # 输出1
match = match_pattern(r'[A-z]', 'b')
print(match)  # 输出b
match = match_pattern(r'[A-z]', '\\')  # [ ]里的符号按照ASCII码进行判断，所以\是匹配的
print(match)  # 输出\

# '*'可以匹配0个或多个前面的字符
match = match_pattern(r'ab*', 'a')
print(match)  # 输出a
match = match_pattern('ab*', 'abbb')
print(match)  # 输出abbb

# '+'可以匹配1个或多个前面的字符
match = match_pattern(r'ab+', 'a')
print(match)  # 输出Fail to match
match = match_pattern(r'ab+', 'abbb')
print(match)  # 输出abbb

# '?'可以匹配0个或1个前面的字符
match = match_pattern(r'ab?', 'a')
print(match)  # 输出a
# '+?'匹配1个前面的字符
match = match_pattern(r'ab+?', 'a')
print(match)  # 输出Faill to match

# '{n}'可以匹配n个前面的字符
match = match_pattern(r'ab{2}', 'ab')
print(match)  # 输出Fail to match
match = match_pattern(r'ab{2}', 'abbb')
print(match)  # 输出abbb

# '{n,m}'可以匹配n到m个前面的字符
match = match_pattern(r'ab{2,3}', 'abbb')
print(match)  # 输出abbb
match = match_pattern(r'ab{2,3}', 'ab')
print(match)  # 输出Fail to match

# '|'表示或的关系
match = match_pattern(r'a[b|c]d', 'ab')
print(match)  # 输出Fail to match
match = match_pattern(r'a[b|c]d', 'acd')
print(match)  # 输出acd

# '()'可以将多个字符组合在一起
match = match_pattern(r'(abc)+?', 'abcabc')
print(match)  # 输出abcabc

# '\'转义符
match = match_pattern(r'\\', '\\')
print(match)  # 输出\
match = match_pattern(r'\$', '$')
print(match)  # 输出$

# '\number' 匹配第一个模式的结果
match = match_pattern(r'([a-z])111\1', 'a111c')  # 首尾相同
print(match)  # 输出Fail to match
match = match_pattern(r'([a-z])111\1', 'a111a')
print(match)  # 输出a111a

# '\b' 匹配单词边界
match = match_pattern(r'\bfoo\b', 'foot')  # 找独立的'foo'
print(match)  # 输出Fail to match
match = match_pattern(r'\bfoo\b', 'foo')
print(match)  # 输出foo

# '\B' 匹配非单词边界
match = match_pattern(r'\Bfoo\B', 'foot')  # 找两边都有字母的'foo'
print(match)  # 输出Fail to match
match = match_pattern(r'\Bfoo\B', 'afootb')
print(match)  # 输出afootb

# '/d' 匹配数字，等价于 '[0-9]'
match = match_pattern(r'\d', '5')
print(match)  # 输出5

# '/D' 匹配非数字，等价于 '[^0-9]'
match = match_pattern(r'\D', '5')
print(match)  # 输出Fail to match

# '/w' 匹配字母数字字符，等价于 '[a-zA-Z0-9_]'
match = match_pattern(r'\w', 'S')
print(match)  # 输出S
match = match_pattern(r'\w', '_')
print(match)  # 输出_

# '/W' 匹配非字母数字字符，等价于 '[^a-zA-Z0-9_]'
match = match_pattern(r'\W', '5')
print(match)  # 输出Fail to match

# '/s'匹配空白字符，等价于 '[\t\n\r\f\v]'
match = match_pattern(r'\s', ' ')
print(match)  # 输出 

# '/S'匹配非空白字符，等价于 '[^\t\n\r\f\v]'
match = match_pattern(r'\S', ' ')
print(match)  # 输出Fail to match

# 事实上这四种(...)可以混合使用，这里仅作四种扩展语法示例
# (?<=p1)p2(?=p3) 匹配p2，前面是p1，后面是p3
match = match_pattern(r'(?<=a)b(?=c)', 'bc')
print(match)  # 输出Fail to match
match = match_pattern(r'(?<=a)b(?=c)', 'abc')
print(match)  # 输出abc

# (?<!p1)p2(?!p3) 匹配p2，前面无p1，后面无p3
match = match_pattern(r'(?<!a)b(?!c)', 'bc')
print(match)  # 输出Fail to match
match = match_pattern(r'(?<!a)b(?!c)', 'ba')
print(match)  # 输出ba

正则表达式的局限

在使用正则表达式时需要注意，它通常只进行形式上的匹配检查，并不保证内容在实际业务中的绝对正确性。正则语法主要关注字符串是否符合预设的字符排列模式，而无法替代业务逻辑层面的深度校验。因此，通过正则验证的数据仍需结合其他规则判断其真实有效性。

# compile()方法可以将正则表达式编译成一个正则表达式对象，供match()和search()使用，使用re.compile()方法能提高效率
# re.compile(pattern, flags=0)
# flags 用 | 组合，常用：
#   re.I   忽略大小写
#   re.L   本地字符集
#   re.M   多行模式（^$匹配每行）
#   re.S   .匹配换行
#   re.U   Unicode（Python3默认）
#   re.X   忽略空格+支持注释

p = re.compile('a[a-z]c')
print(p.match('abc').group())  # 输出abc
print(p.search('abc').group())  # 输出abc

# re.match()方法从字符串的起始位置开始匹配，如果匹配成功则返回匹配对象，否则返回None
res01 = re.match('^python', 'python is a programming language')
res02 = re.match('python$', 'this is python')
print(res01)  # 输出<_sre.SRE_Match object; span=(0, 6), match='python'>
print(res01.group())  # 输出python
print(res02)  # 输出None

# re.search()方法从字符串的任意位置开始匹配，如果匹配成功则返回匹配对象，否则返回None
res01 = re.search('python', 'Python is a programming language', re.I)
res02 = re.search('python', 'this is Python', re.I)
res03 = re.search('python', 'this is Python?', re.I)
print(res01)  # 输出<_sre.SRE_Match object; span=(0, 6), match='Python'>
print(res02)  # 输出<_sre.SRE_Match object; span=(8, 14), match='Python'>
print(res03)  # 输出<_sre.SRE_Match object; span=(8, 14), match='Python'>

# 使用re模块来匹配对象
res01 = re.match('^python', 'python is a programming language')
if res01:  # 如果匹配失败，res01为None，None对象为假
    print(res01.group())  # 输出python
else:
    print('no match')

# group()方法返回匹配对象中分组匹配的字符串
res01 = re.match(r'(\w+) (\w+)', 'Say Hello to My Little Friend')
print(res01.group(0)) # 输出Say Hello
print(res01.group(1)) # 输出Say
print(res01.group(2)) # 输出Hello
print(res01.group(1, 2)) # 输出('Say', 'Hello')

# re.findall()方法返回所有匹配的字符串，返回的是一个列表
res01 = re.findall(r'\d+', 'abc123cdf456')
print(res01) # 输出['123', '456']
res02 = re.findall(r'\d+', 'abccdf') # 没有匹配到返回空列表
print(res02) # 输出[]

# re.split()方法将字符串按照匹配的子串分割后返回列表
res01 = re.split(r'\s+', 'I hear the wind from the landscape.')
print(res01) # 输出['I', 'hear', 'the', 'wind', 'from', 'the', 'landscape.']
res02 = re.split(r'\s+', 'I hear the wind from the landscape.', maxsplit = 3) # maxsplit指定最大分割次数
print(res02) # 输出['I', 'hear', 'the', 'wind from the landscape.']

# re.sub()方法将字符串中的匹配子串用字符串替换，返回替换后的字符串
res01 = re.sub(r'\s+', '! ', 'My Little Pony ')
print(res01) # 输出My! Little! Pony!
res02 = re.sub(r'\s+', '! ', 'My Little Pony ', count = 2) # count指定替换的次数
print(res02) # 输出My! Little! Pony

# re.subn()方法与re.sub()方法类似，返回替换后的字符串和替换的次数
res01 = re.subn(r'\s+', '! ', 'My Little Pony ')
print(res01) # 输出('My! Little! Pony!', 3)
