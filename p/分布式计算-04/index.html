分布式线性回归

实现思路

前提：n » p，且 p 不太大时，\( X^T X \) 和 \( X^T Y \) 可以装入内存。

Step 1：分块 将原始数据按行分成 k 个分块，每个分块包含对应的 X 和 Y（保持 X 和 Y 在同一分块内）。

Step 2：分布式计算 每个节点在本地计算 \( X_i^T X_i \) 和 \( X_i^T Y_i \)，然后将结果传回主节点。

Step 3：汇总求解 主节点将所有分块结果累加：

\[ X^T X = \sum_{i=1}^k X_i^T X_i, \quad X^T Y = \sum_{i=1}^k X_i^T Y_i \]

最后解线性方程组 \((X^T X)\hat{\beta} = X^T Y\) 得到 OLS 估计值。

import ray
import numpy as np
import pandas as pd
import scipy
import time
import os

ray.init()

file_name = "ols_data.csv"
n_obs = 50000  # 5万个观测
p = 100         # 100个变量

# 设定一个真实的 beta（包含截距项，共101个参数）
np.random.seed(123)
beta_true = np.random.uniform(-5, 5, size=p + 1)

print(f"正在生成 {n_obs} 行测试数据并写入 CSV...")
t0 = time.time()

# 分块写入数据以防我们在生成数据时超出内存限制
chunk_size_gen = 5000
if os.path.exists(file_name):
    os.remove(file_name)

for i in range(n_obs // chunk_size_gen):
    X_chunk = np.random.randn(chunk_size_gen, p)
    # 计算 Y = beta0 + X * beta + noise
    noise = np.random.randn(chunk_size_gen) * 2.0
    y_chunk = beta_true[0] + X_chunk @ beta_true[1:] + noise

    # 拼接并写入
    data_chunk = np.column_stack((X_chunk, y_chunk))
    df_chunk = pd.DataFrame(data_chunk)
    df_chunk.to_csv(file_name, mode="a", index=False, header=False)

t1 = time.time()
print(f"数据生成完毕！耗时: {t1 - t0:.2f} 秒。文件大小约为：{os.path.getsize(file_name)/(1024**2):.3f} MB")

@ray.remote
def compute_ols_stats(chunk_data):
    # chunk_data 是一个 numpy 矩阵，最后一列是 Y，前面是 X
    X_chunk = chunk_data[:, :-1]
    y_chunk = chunk_data[:, -1]
    n_chunk = X_chunk.shape[0]

    # 在设计矩阵 X 前添加一列1向量，用于拟合截距项
    intercept_col = np.ones((n_chunk, 1))
    X_chunk_with_intercept = np.hstack((intercept_col, X_chunk))

    # 计算局部充分统计量
    XtX_partial = X_chunk_with_intercept.T @ X_chunk_with_intercept
    Xty_partial = X_chunk_with_intercept.T @ y_chunk

    return XtX_partial, Xty_partial

chunk_size = 5000
tasks = []

print("开始流式读取并分发 OLS 计算任务...")
t0 = time.time()

for i, chunk in enumerate(pd.read_csv(file_name, header=None, chunksize=chunk_size)):
    # 将局部数据送入共享内存
    chunk_id = ray.put(chunk.values)

    # 派发计算任务
    task = compute_ols_stats.remote(chunk_id)
    tasks.append(task)
    print(f"  已派发第 {i+1} 块数据...")

print("数据分发完毕，等待集群计算...")

results = ray.get(tasks)

# results 是一个列表，里面包含了每个任务返回的元组 (XtX_partial, Xty_partial)
XtX_global = sum([res[0] for res in results])  # 这里用sum函数，不用np.sum
Xty_global = sum([res[1] for res in results])

print("正在通过 Cholesky 分解求解 beta_hat...")

cho_fac = scipy.linalg.cho_factor(XtX_global, lower=True)
beta_hat = scipy.linalg.cho_solve(cho_fac, Xty_global)

# 对比前10个参数（包含截距）
print("\n=== 参数对比 ===")
for i in range(10):
    print(f"参数{i}：真实值 = {beta_true[i]:>7.4f}, 估计值 = {beta_hat[i]:>7.4f}")
